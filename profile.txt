const mongoose = require("mongoose");

const userSchema = new mongoose.Schema(
    {
        name: {
            type: String,
            required: [true, "Name is required"],
            trim: true,
            minlength: 2,
            maxlength: 60,
        },
        email: {
            type: String,
            required: [true, "Email is required"],
            unique: true,
            lowercase: true,
            trim: true,
        },
        phone: {
            type: String,
            trim: true,
        },
        passwordHash: {
            type: String,
            required: true,
            select: false, // never send the password back in API responses
        },
        role: {
            type: String,
            enum: ["student", "admin"],
            default: "student",
        },
        isEmailVerified: {
            type: Boolean,
            default: false,
        },
        status: {
            type: String,
            enum: ["active", "suspended"],
            default: "active",
        },
        streak: {
            current: { type: Number, default: 0 },
            longest: { type: Number, default: 0 },
            lastActiveDate: { type: Date },
        },
        lastLoginAt: { type: Date },
    },
    { timestamps: true } // adds createdAt and updatedAt automatically
);

module.exports = mongoose.model("User", userSchema);
===== server\controllers\authController.js =====
const bcrypt = require("bcryptjs");
const jwt = require("jsonwebtoken");
const User = require("../models/User");

// Creates a login token that expires in 7 days
const createToken = (userId, role) => {
    return jwt.sign({ id: userId, role }, process.env.JWT_SECRET, {
        expiresIn: "7d",
    });
};

// Shapes the user data we send back (never the password)
const publicUser = (user) => ({
    id: user._id,
    name: user.name,
    email: user.email,
    role: user.role,
});

// POST /api/auth/register
const register = async (req, res) => {
    try {
        const { name, email, password, phone } = req.body;

        if (!name || !email || !password) {
            return res
                .status(400)
                .json({ message: "Name, email and password are required" });
        }

        if (password.length < 6) {
            return res
                .status(400)
                .json({ message: "Password must be at least 6 characters" });
        }

        const existingUser = await User.findOne({ email: email.toLowerCase() });
        if (existingUser) {
            return res.status(409).json({ message: "Email is already registered" });
        }

        const passwordHash = await bcrypt.hash(password, 10);
        const user = await User.create({ name, email, phone, passwordHash });

        res.status(201).json({
            message: "Registration successful",
            token: createToken(user._id, user.role),
            user: publicUser(user),
        });
    } catch (error) {
        console.error("Register error:", error.message);
        res.status(500).json({ message: "Something went wrong on the server" });
    }
};

// POST /api/auth/login
const login = async (req, res) => {
    try {
        const { email, password } = req.body;

        // 1. Check both fields are filled
        if (!email || !password) {
            return res
                .status(400)
                .json({ message: "Email and password are required" });
        }

        // 2. Find the user. "+passwordHash" asks for the hidden password field.
        const user = await User.findOne({ email: email.toLowerCase() }).select(
            "+passwordHash"
        );

        // 3. Compare the typed password with the scrambled one
        const passwordMatches =
            user && (await bcrypt.compare(password, user.passwordHash));

        // Same message for both cases, so attackers can't tell which emails exist
        if (!passwordMatches) {
            return res.status(401).json({ message: "Invalid email or password" });
        }

        // 4. Block suspended accounts
        if (user.status !== "active") {
            return res
                .status(403)
                .json({ message: "Your account is suspended. Contact support." });
        }

        // 5. Remember when they last logged in
        user.lastLoginAt = new Date();
        await user.save();

        res.json({
            message: "Login successful",
            token: createToken(user._id, user.role),
            user: publicUser(user),
        });
    } catch (error) {
        console.error("Login error:", error.message);
        res.status(500).json({ message: "Something went wrong on the server" });
    }
};

// GET /api/auth/me (only for logged-in users)
const getMe = async (req, res) => {
    // req.user was filled in by the auth middleware
    res.json({ user: publicUser(req.user) });
};

module.exports = { register, login, getMe };

===== client\src\context\AuthContext.jsx =====
import { createContext, useContext, useEffect, useState } from "react";
import api from "../api/axios";

const AuthContext = createContext(null);

export function AuthProvider({ children }) {
    const [user, setUser] = useState(null);
    const [loading, setLoading] = useState(true); // true while we check a saved token

    // When the app opens: if a token was saved, ask the backend who it belongs to
    useEffect(() => {
        const token = localStorage.getItem("token");
        if (!token) {
            setLoading(false);
            return;
        }
        api
            .get("/auth/me")
            .then((res) => setUser(res.data.user))
            .catch(() => localStorage.removeItem("token")) // expired or invalid token
            .finally(() => setLoading(false));
    }, []);

    // Save the token so the user stays logged in after refreshing
    const saveSession = (data) => {
        localStorage.setItem("token", data.token);
        setUser(data.user);
    };

    const login = async (email, password) => {
        const res = await api.post("/auth/login", { email, password });
        saveSession(res.data);
        return res.data.user;
    };

    const register = async (name, email, password) => {
        const res = await api.post("/auth/register", { name, email, password });
        saveSession(res.data);
        return res.data.user;
    };

    const logout = () => {
        localStorage.removeItem("token");
        setUser(null);
    };

    return (
        <AuthContext.Provider value={{ user, loading, login, register, logout }}>
            {children}
        </AuthContext.Provider>
    );
}

// Any page can call useAuth() to get the user, login, logout, etc.
export const useAuth = () => useContext(AuthContext);

import { Link, NavLink, Navigate, Outlet } from "react-router-dom";
import { choiceKey } from "../pages/student/Welcome";
import { useAuth } from "../context/AuthContext";
import { ExamProvider, useExam } from "../context/ExamContext";

const LINKS = [
  { to: "/dashboard", label: "Dashboard", icon: "M3 11l9-7 9 7M5 10v10h5v-6h4v6h5V10" },
  { to: "/practice", label: "Practice", icon: "M5 4h11l3 3v13H5zM9 10h6M9 14h6" },
  { to: "/mock-tests", label: "Mock tests", icon: "M12 7v5l3 2M12 3a9 9 0 100 18 9 9 0 000-18z" },
  { to: "/analytics", label: "Analytics", icon: "M4 20V10M10 20V4M16 20v-6M2 20h20" },
  { to: "/revise", label: "Revise", icon: "M4 4v6h6M20 20v-6h-6M5.5 15a7.5 7.5 0 0013.5 2M18.5 9A7.5 7.5 0 005 7" },
];

function Icon({ path }) {
  return (
    <svg viewBox="0 0 24 24" className="w-5 h-5" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" aria-hidden="true">
      <path d={path} />
    </svg>
  );
}

function ExamPicker() {
  const { exams, examId, setExamId } = useExam();
  if (exams.length === 0) return null;
  return (
    <select
      value={examId}
      onChange={(e) => setExamId(e.target.value)}
      aria-label="Exam"
      className="rounded-lg border border-slate-300 bg-white px-2 py-1.5 text-sm font-semibold text-indigo-700 focus:outline-none focus:ring-2 focus:ring-indigo-500"
    >
      {exams.map((e) => (
        <option key={e._id} value={e._id}>
          {e.code}
        </option>
      ))}
    </select>
  );
}

function Shell() {
  const { user, logout } = useAuth();
  const { examId } = useExam();
  if (!localStorage.getItem(choiceKey(user))) return <Navigate to="/welcome" replace />;

  const linkClass = ({ isActive }) =>
    `px-3 py-2 rounded-lg text-sm font-medium ${
      isActive ? "bg-indigo-50 text-indigo-700" : "text-slate-600 hover:text-slate-900 hover:bg-slate-100"
    }`;
  const tabClass = ({ isActive }) =>
    `flex-1 flex flex-col items-center gap-0.5 py-2 text-xs font-medium ${isActive ? "text-indigo-700" : "text-slate-500"}`;

  return (
    <div className="min-h-screen bg-slate-50">
      <header className="bg-white border-b border-slate-200 sticky top-0 z-20">
        <div className="max-w-6xl mx-auto px-4 h-14 flex items-center gap-4">
          <Link to="/dashboard" className="flex items-center gap-2 font-bold text-slate-900">
            <img src="/favicon.svg" alt="" className="w-7 h-7" />
            <span>ExamPrep AI</span>
          </Link>

          <nav className="hidden md:flex items-center gap-1 ml-4" aria-label="Main">
            {LINKS.map((l) => (
              <NavLink key={l.to} to={l.to} className={linkClass}>
                {l.label}
              </NavLink>
            ))}
            {user.role === "admin" && (
              <NavLink to="/admin" className={linkClass}>
                Admin
              </NavLink>
            )}
          </nav>

          <div className="ml-auto flex items-center gap-3">
            <ExamPicker />
            <span className="hidden sm:block text-sm text-slate-500 max-w-32 truncate">{user.name}</span>
            <button
              onClick={logout}
              className="text-sm font-medium text-slate-600 hover:text-slate-900 border border-slate-300 rounded-lg px-3 py-1.5"
            >
              Log out
            </button>
          </div>
        </div>
      </header>

      <main className="pb-20 md:pb-0">
        {/* key: when the exam changes, the page reloads its data for the new exam */}
        <Outlet key={examId} />
      </main>

      <nav className="md:hidden fixed bottom-0 inset-x-0 z-20 bg-white border-t border-slate-200 flex" aria-label="Main">
        {LINKS.map((l) => (
          <NavLink key={l.to} to={l.to} className={tabClass}>
            <Icon path={l.icon} />
            {l.label}
          </NavLink>

      </nav>
